```python from flask import Flask, request, redirect, url_for, send_from_directory, render_template_string import os from werkzeug.utils import secure_filename app = Flask(__name__) # Cloud storage folder CLOUD_FOLDER = "cloud_storage" RESTORE_FOLDER = "restored_files" os.makedirs(CLOUD_FOLDER, exist_ok=True) os.makedirs(RESTORE_FOLDER, exist_ok=True) app.config["CLOUD_FOLDER"] = CLOUD_FOLDER HTML = """ Cloud Backup & Restore System

☁ Cloud-Based Backup & Restore System

Securely backup and restore your files

{% if message %}
{{ message }}
{% endif %}

📤 Backup File



☁ Cloud Backup Storage

{% if files %} {% for file in files %} {% endfor %}
File Name Size Actions
{{ file.name }} {{ file.size }} KB Restore Download Delete
{% else %}

No backup files available.

{% endif %}
Cloud Computing Mini Project
""" def get_files(): files = [] for filename in os.listdir(CLOUD_FOLDER): filepath = os.path.join(CLOUD_FOLDER, filename) if os.path.isfile(filepath): size = round(os.path.getsize(filepath) / 1024, 2) files.append({ "name": filename, "size": size }) return files @app.route("/") def home(): files = get_files() return render_template_string( HTML, files=files, message=None ) @app.route("/backup", methods=["POST"]) def backup(): if "file" not in request.files: return redirect(url_for("home")) file = request.files["file"] if file.filename == "": return redirect(url_for("home")) filename = secure_filename(file.filename) filepath = os.path.join( CLOUD_FOLDER, filename ) file.save(filepath) files = get_files() return render_template_string( HTML, files=files, message="File backed up successfully to cloud storage!" ) @app.route("/restore/") def restore(filename): filename = secure_filename(filename) source = os.path.join( CLOUD_FOLDER, filename ) destination = os.path.join( RESTORE_FOLDER, filename ) if os.path.exists(source): with open(source, "rb") as src: data = src.read() with open(destination, "wb") as dest: dest.write(data) files = get_files() return render_template_string( HTML, files=files, message="File restored successfully to the restored_files folder!" ) return redirect(url_for("home")) @app.route("/download/") def download(filename): filename = secure_filename(filename) return send_from_directory( CLOUD_FOLDER, filename, as_attachment=True ) @app.route("/delete/") def delete(filename): filename = secure_filename(filename) filepath = os.path.join( CLOUD_FOLDER, filename ) if os.path.exists(filepath): os.remove(filepath) return redirect(url_for("home")) if __name__ == "__main__": print("--------------------------------------") print(" Cloud Backup & Restore System") print("--------------------------------------") print("Server running at:") print("http://127.0.0.1:5000") print("--------------------------------------") app.run(debug=True) ```
